
DO $$
DECLARE
	age int := 18;
BEGIN
	IF age <= 18 THEN
		RAISE NOTICE 'you are % years old and not eligible', age;
	END IF;
END;
$$ LANGUAGE plpgsql;

DO $$
DECLARE
	val int := 4;
BEGIN
	IF val > 0 AND val < 10 THEN
		RAISE NOTICE 'num % is btw 1 and 10', val;
	ELSIF val > 10 AND val < 20 THEN
		RAISE NOTICE 'num % is btw 10 and 20', val;
	ELSE
		RAISE NOTICE 'num % is more than 20', val;
	END IF;
END;
$$ LANGUAGE plpgsql;
